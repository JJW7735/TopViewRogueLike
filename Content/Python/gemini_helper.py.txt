import subprocess
import unreal

def ask_gemini(prompt):
    # CLI 실행 명령어 (실제 설치하신 제미나이 CLI 명령어에 맞게 'gemini-cli' 부분을 수정하세요)
    command = ["gemini-cli", prompt]
    
    try:
        # CLI를 백그라운드에서 실행하고 결과값을 캡처합니다.
        result = subprocess.run(command, capture_output=True, text=True, check=True, encoding='utf-8')
        
        # 앞뒤 공백을 자르고 텍스트 추출
        response = result.stdout.strip()
        
        # 언리얼 엔진의 '출력 로그(Output Log)' 창에 결과 출력
        unreal.log(f"Gemini Response:\n{response}")
        return response
        
    except subprocess.CalledProcessError as e:
        # CLI 실행 중 에러가 발생했을 때의 처리
        unreal.log_error(f"Gemini CLI Error:\n{e.stderr}")
        return None
